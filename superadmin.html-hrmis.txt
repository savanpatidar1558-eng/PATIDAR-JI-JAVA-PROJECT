  <script>
    document.addEventListener("DOMContentLoaded", function() {
      document.getElementById('dateFilter').valueAsDate = new Date();
      
      // Set current month for PDF generator
      let now = new Date();
      let monthStr = now.getFullYear() + "-" + String(now.getMonth() + 1).padStart(2, '0');
      document.getElementById('pdfMonthSelect').value = monthStr;

      loadDashboardData();
    });

    function loadDashboardData() {
      document.getElementById('loader').style.display = 'flex';
      var selectedDate = document.getElementById('dateFilter').value;
      
      google.script.run.withSuccessHandler(function(res) {
        google.script.run.withSuccessHandler(function(extraData) {
          document.getElementById('loader').style.display = 'none';
          
          // KPI & Finance
          document.getElementById('kpiTotal').innerText = res.totalActive;
          document.getElementById('kpiPresent').innerText = res.stats.present;
          document.getElementById('kpiAbsent').innerText = res.stats.absent;
          document.getElementById('kpiLeaves').innerText = res.leaves.length;
          if(res.finance) {
            document.getElementById('kpiSalary').innerText = "₹" + res.finance.salary;
            document.getElementById('kpiAdvance').innerText = "₹" + res.finance.advance;
          }

          // 📸 CCTV
          let selfieHtml = "";
          if(extraData.selfies && extraData.selfies.length > 0) {
              extraData.selfies.forEach(s => {
                  selfieHtml += `<div class="selfie-card"><div class="live-dot"></div><img src="${s.url}" alt="Selfie" onerror="this.src='https://via.placeholder.com/150?text=No+Photo'"><div class="info"><b>${s.name}</b>🕒 ${s.time}</div></div>`;
              });
          } else { selfieHtml = `<div style="color:#888; grid-column: 1 / -1; text-align:center; padding: 20px;">अभी तक कैमरे में कोई डाटा नहीं आया है</div>`; }
          document.getElementById('selfieContainer').innerHTML = selfieHtml;

          // 🗂️ Locker & PDF Staff Dropdown
          let lockerHtml = "";
          let staffOptions = "<option value=''>स्टाफ चुनें...</option>";
          
          if(extraData.locker && extraData.locker.length > 0) {
            extraData.locker.forEach(staff => {
              staffOptions += `<option value="${staff.name}">${staff.name}</option>`;
              let firstLetter = staff.name.charAt(0).toUpperCase();
              let docBtn = staff.docLink && staff.docLink.includes("http") ? `<a href="${staff.docLink}" target="_blank" class="doc-btn">📄 View Document</a>` : `<span style="color:#e74c3c; font-size:11px; display:block; margin-top:10px;">❌ No Document</span>`;
              lockerHtml += `<div class="locker-card"><div class="avatar">${firstLetter}</div><h4>${staff.name}</h4><p style="color:var(--primary);">${staff.role}</p><p>📱 ${staff.mobile}</p><p>💰 ₹${staff.salary}</p>${docBtn}</div>`;
            });
          }
          document.getElementById('lockerContainer').innerHTML = lockerHtml;
          document.getElementById('pdfStaffSelect').innerHTML = staffOptions;

          // 📍 Maps
          let attHtml = "";
          res.stats.list.forEach(s => {
            let color = s.status.includes("Present") ? "#27ae60" : (s.status.includes("Half Day") || s.status.includes("Late") ? "#f39c12" : "#e74c3c");
            let locString = extraData.locations[s.name];
            let mapBtn = (locString && locString !== "" && s.inTime !== "--:--") ? `<br><a href="https://maps.google.com/?q=$${encodeURIComponent(locString)}" target="_blank" class="map-btn">📍 View Map</a>` : "";
            attHtml += `<tr><td><b>${s.name}</b></td><td>${s.inTime} ${mapBtn}</td><td style="color:${color}; font-weight:bold;">${s.status}</td></tr>`;
          });
          document.getElementById('attTableBody').innerHTML = attHtml || `<tr><td colspan="3" style="text-align:center;">आज कोई डेटा नहीं है</td></tr>`;

          // Leaves
          let leaveHtml = "";
          if(res.leaves.length > 0) {
            res.leaves.forEach(l => { leaveHtml += `<tr><td><b>${l.name}</b></td><td><span style="color:#27ae60;">${l.leaveDate}</span><br><small>${l.reason}</small></td><td><button class="btn btn-approve" onclick="handleLeave(${l.row}, 'Approve', '${l.name}')">✅</button> <button class="btn btn-reject" onclick="handleLeave(${l.row}, 'Reject', '${l.name}')">❌</button></td></tr>`; });
          } else { leaveHtml = `<tr><td colspan="3" style="text-align:center; padding:15px; color:#888;">कोई पेंडिंग रिक्वेस्ट नहीं 🎉</td></tr>`; }
          document.getElementById('leaveTableBody').innerHTML = leaveHtml;

        }).getMapAndLockerData(selectedDate);
      }).getSuperAdminAnalytics(selectedDate);
    }

    // 📑 PDF Generator Function
    function downloadMonthlyPDF() {
      let staff = document.getElementById('pdfStaffSelect').value;
      let month = document.getElementById('pdfMonthSelect').value;
      
      if(!staff || !month) return alert("❌ कृपया स्टाफ का नाम और महीना चुनें!");
      
      document.getElementById('loader-text').innerText = "PDF जनरेट हो रहा है... (इसमें 5-10 सेकंड लग सकते हैं)";
      document.getElementById('loader').style.display = 'flex';
      
      google.script.run.withSuccessHandler(function(res) {
        document.getElementById('loader').style.display = 'none';
        if(res.error) return alert(res.error);
        window.open(res.url, "_blank");
      }).generateStaffMonthlyAttendancePDF(staff, month);
    }

    function handleLeave(row, action, name) {
      let actionText = action === 'Approve' ? 'मंज़ूर (Approve)' : 'रद्द (Reject)';
      let remark = prompt(`आप ${name} की छुट्टी ${actionText} कर रहे हैं। \nरिमार्क (Remark):`, "OK");
      if(remark !== null) {
        document.getElementById('loader-text').innerText = `प्रोसेसिंग...`;
        document.getElementById('loader').style.display = 'flex';
        google.script.run.withSuccessHandler(function() {
          alert(`✅ छुट्टी ${actionText} हो गई!`); loadDashboardData(); 
        }).processLeave(row, action, remark, name);
      }
    }

    function publishNotice() {
      var msg = document.getElementById("noticeMsg").value;
      if(!msg) return alert("❌ कुछ मैसेज लिखें!");
      google.script.run.withSuccessHandler(function(res) {
        alert(res); document.getElementById("noticeMsg").value = "";
      }).saveNotice(msg);
    }
